Public Sub StartGame()

    CancelTimer

    UnbindArrowKeys

    gameRunning = False
    gamePaused = False

    Randomize

    'The chilled stadium music is on by default and each
    'new game starts with a random track (shuffle).
    If musicInitialized = False Then
        musicOn = True
        musicInitialized = True
    End If

    currentMusicTrack = Int(Rnd * 3) + 1

    'Music plays from the team selection screen onwards.
    StartMusic

    '=====================================================
    ' SPEED: the selection screen and the level are built
    ' while the screen is hidden and shown in one repaint.
    '=====================================================
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False

    If SelectTeam() = False Then

        StopMusic
        CloseSound

        Application.Calculation = xlCalculationAutomatic
        Application.EnableEvents = True
        Application.ScreenUpdating = True

        Exit Sub

    End If

    'SelectTeam re-enables the screen so the flags can be
    'clicked - hide it again for the fast level build.
    Application.ScreenUpdating = False

    currentLevel = 1

    score = 0
    goalsCollected = 0

    levelStartScore = 0
    levelStartGoals = 0

    gameRunning = True
    gamePaused = False

    totalPausedSeconds = 0
    gameStart = Now

    'Force the static dashboard cells to be rewritten.
    lastStageKey = ""

    'SPEED: the dashboard itself is built inside StartLevel
    'while the screen is hidden, so it is only created once.
    SetupHighScores

    BindArrowKeys

    StartLevel

    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    Application.ScreenUpdating = True

End Sub


Private Function SelectTeam() As Boolean

    Dim ws As Worksheet
    Dim teams As Variant
    Dim i As Long
    Dim shapeIndex As Long
    Dim shp As Shape
    Dim startLeft As Double
    Dim startTop As Double
    Dim flagWidth As Double
    Dim flagHeight As Double
    Dim gapX As Double
    Dim gapY As Double
    Dim colIndex As Long
    Dim rowIndex As Long

    Set ws = ThisWorkbook.Worksheets(SHEET_NAME)

    teams = Array( _
        "Algeria", "Argentina", "Australia", "Austria", _
        "Belgium", "Bosnia and Herzegovina", "Brazil", "Cabo Verde", _
        "Canada", "Colombia", "Congo DR", "Croatia", _
        "Curaçao", "Czechia", "Ecuador", "Egypt", _
        "England", "France", "Germany", "Ghana", _
        "Haiti", "IR Iran", "Iraq", "Japan", _
        "Jordan", "Korea Republic", "Mexico", "Morocco", _
        "Netherlands", "New Zealand", "Norway", "Panama", _
        "Paraguay", "Portugal", "Qatar", "Saudi Arabia", _
        "Scotland", "Senegal", "South Africa", "Spain", _
        "Sweden", "Switzerland", "Tunisia", "Türkiye", _
        "USA", "Uruguay", "Uzbekistan", "Côte d'Ivoire")

    teamSelectionMade = False
    selectedTeam = ""
    selectedTeamFlag = ""

    '=====================================================
    ' SPEED: build the entire selection screen while the
    ' screen is hidden and show it in one repaint.
    '=====================================================
    Application.ScreenUpdating = False

    On Error Resume Next

    ws.Range("B1:X2").UnMerge
    ws.Range("B3:X3").UnMerge

    For Each shp In ws.Shapes

        If Left$(shp.Name, 8) = "WCFlag__" _
            Or Left$(shp.Name, 14) = "WCFlagVisual__" _
            Or Left$(shp.Name, 14) = "WCTrophyAnim__" Then
            shp.Delete
        End If

    Next shp

    On Error GoTo 0

    'Cleared down to row 40 so no part of a finished
    'level 5 maze (which reaches row 26) is left behind.
    ws.Range("B1:X40").Clear

    'World Cup stadium background for the selection screen.
    PaintStadiumBackground ws

    'Keep the high score table styled correctly.
    RestoreHighScoreFills ws

    With ws.Range("B1:X2")

        .Merge

        .value = WorldCupSymbol("TROPHY") & _
            "  SELECT YOUR TEAM  " & _
            WorldCupSymbol("TROPHY")

        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter

        .Font.Name = "Segoe UI Emoji"
        .Font.Bold = True
        .Font.Size = 24

        .Font.Color = RGB(255, 215, 0)

        .Interior.Color = RGB(15, 35, 60)

        .Borders.LineStyle = xlContinuous
        .Borders.Color = RGB(255, 215, 0)
        .Borders.Weight = xlMedium

    End With

    With ws.Range("B3:X3")

        .Merge

        .value = "CLICK A FLAG TO REPRESENT YOUR COUNTRY"

        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter

        .Font.Bold = True
        .Font.Size = 12

        .Font.Color = RGB(255, 255, 255)

        .Interior.Color = RGB(35, 75, 55)

        .Borders.LineStyle = xlContinuous
        .Borders.Color = RGB(255, 255, 255)

    End With

    '48 flags arranged in the same compact 8 x 6 grid.
    startLeft = ws.Range("B5").Left
    startTop = ws.Range("B5").top

    flagWidth = 52
    flagHeight = 34
    gapX = 58
    gapY = 39

    For i = LBound(teams) To UBound(teams)

        colIndex = i Mod 8
        rowIndex = i \ 8

        shapeIndex = i + 1

        'Draw the actual flag graphic.
        DrawTeamFlagGraphic _
            ws, _
            i, _
            startLeft + (colIndex * gapX), _
            startTop + (rowIndex * gapY), _
            flagWidth, _
            flagHeight, _
            "WCFlagVisual__" & Format$(shapeIndex, "00")

        'Transparent click target over the flag.
        Set shp = ws.Shapes.AddShape( _
            msoShapeRectangle, _
            startLeft + (colIndex * gapX), _
            startTop + (rowIndex * gapY), _
            flagWidth, _
            flagHeight)

        shp.Name = "WCFlag__" & Format$(shapeIndex, "00")

        shp.Fill.Transparency = 1
        shp.Line.Transparency = 1

        shp.AlternativeText = teams(i)
        shp.OnAction = "'" & ThisWorkbook.Name & "'!SelectTeamByFlag"

    Next i

    'Show the finished selection screen.
    Application.ScreenUpdating = True

    Do While teamSelectionMade = False

        DoEvents

    Loop

    On Error Resume Next

    For Each shp In ws.Shapes

        If Left$(shp.Name, 8) = "WCFlag__" _
            Or Left$(shp.Name, 14) = "WCFlagVisual__" _
            Or Left$(shp.Name, 14) = "WCTrophyAnim__" Then
            shp.Delete
        End If

    Next shp

    On Error GoTo 0

    SelectTeam = True

End Function

Private Sub DrawFlagUnionJack( _
    ByVal ws As Worksheet, _
    ByVal cjLeft As Double, _
    ByVal cjTop As Double, _
    ByVal cjW As Double, _
    ByVal cjH As Double, _
    ByVal namePrefix As String)

    Dim shp As Shape
    Dim diagAngle As Double
    Dim diagLen As Double

    diagAngle = Atn(cjH / cjW) * 180 / (Atn(1) * 4)
    diagLen = Sqr(cjW * cjW + cjH * cjH) * 1.3

    AddFlagRect ws, cjLeft, cjTop, cjW, cjH, RGB(0, 36, 125), namePrefix & "__CJBASE"

    Set shp = ws.Shapes.AddShape(msoShapeRectangle, _
        cjLeft + cjW / 2 - diagLen / 2, _
        cjTop + cjH / 2 - cjH * 0.15, _
        diagLen, cjH * 0.3)
    shp.Name = namePrefix & "__CJD1"
    shp.Fill.ForeColor.RGB = RGB(255, 255, 255)
    shp.Line.Visible = msoFalse
    shp.rotation = diagAngle

    Set shp = ws.Shapes.AddShape(msoShapeRectangle, _
        cjLeft + cjW / 2 - diagLen / 2, _
        cjTop + cjH / 2 - cjH * 0.15, _
        diagLen, cjH * 0.3)
    shp.Name = namePrefix & "__CJD2"
    shp.Fill.ForeColor.RGB = RGB(255, 255, 255)
    shp.Line.Visible = msoFalse
    shp.rotation = -diagAngle

    Set shp = ws.Shapes.AddShape(msoShapeRectangle, _
        cjLeft + cjW / 2 - diagLen / 2, _
        cjTop + cjH / 2 - cjH * 0.07, _
        diagLen, cjH * 0.14)
    shp.Name = namePrefix & "__CJD3"
    shp.Fill.ForeColor.RGB = RGB(206, 17, 38)

    Dim secondLetter As Long

        Case 7: firstLetter = 2: secondLetter = 21
        Case 8: firstLetter = 2: secondLetter = 0
        Case 9: firstLetter = 2: secondLetter = 14
        Case 10: firstLetter = 2: secondLetter = 3
        Case 11: firstLetter = 7: secondLetter = 17
        Case 12: firstLetter = 2: secondLetter = 22
        Case 13: firstLetter = 2: secondLetter = 25
        Case 14: firstLetter = 4: secondLetter = 2
        Case 15: firstLetter = 4: secondLetter = 6
        Case 16: firstLetter = 6: secondLetter = 1
        Case 17: firstLetter = 5: secondLetter = 17
        Case 18: firstLetter = 3: secondLetter = 4
        Case 19: firstLetter = 6: secondLetter = 7
        Case 20: firstLetter = 7: secondLetter = 19
        Case 21: firstLetter = 8: secondLetter = 17
        Case 22: firstLetter = 8: secondLetter = 16
        Case 23: firstLetter = 9: secondLetter = 15
        Case 24: firstLetter = 9: secondLetter = 14
        Case 25: firstLetter = 10: secondLetter = 17
        Case 26: firstLetter = 12: secondLetter = 23
        Case 27: firstLetter = 12: secondLetter = 0

'=========================================================
' START LEVEL
'=========================================================

Private Sub StartLevel()

    Dim oldScreenUpdating As Boolean

    oldScreenUpdating = Application.ScreenUpdating

    '=========================================================
    ' SPEED OPTIMISATION
    '
    ' The maze is drawn one cell at a time. Turning screen
    ' updating off while the level is being created prevents
    ' Excel from repainting the worksheet after every cell.
    ' The finished maze is then shown all at once.
    '=========================================================
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False

    On Error GoTo RestoreScreenAndRaise

    CancelTimer

    'Save checkpoint for this level
    levelStartScore = score
    levelStartGoals = goalsCollected

    lives = LivesForLevel()

    gameRunning = True
    gamePaused = False

    'Bring back the chilled music for the new level.
    StartMusic

    currentMazeSize = MazeSizeForLevel()

    SetupDashboard

    If BuildMaze() = False Then

        Application.Calculation = xlCalculationAutomatic
        Application.EnableEvents = True
        Application.ScreenUpdating = oldScreenUpdating

        gameRunning = False

        UnbindArrowKeys

        MsgBox "The maze could not be created." & vbCrLf & vbCrLf & _
            "Please try starting the game again.", _
            vbCritical, _
            "World Cup Escape"

        Exit Sub

    End If

    levelStartElapsedSeconds = GetElapsedSeconds()

    UpdateDashboard

    'Draw the team flag once per level (not on every move).
    RefreshDashboardTeamFlag

    SetStatus "PLAYING"

    UpdateTimer

    'Show the completed maze only after all drawing is finished.
    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    Application.ScreenUpdating = oldScreenUpdating

    Exit Sub

RestoreScreenAndRaise:

    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    Application.ScreenUpdating = oldScreenUpdating

    Err.Raise Err.Number, Err.Source, Err.Description

End Sub

'=========================================================
' BUILD MAZE
'=========================================================

Private Function BuildMaze() As Boolean

    Dim ws As Worksheet
    Dim bgRow As Long

    Set ws = ThisWorkbook.Worksheets(SHEET_NAME)

    BuildMaze = False

    currentMazeSize = MazeSizeForLevel()

    'Clear previous maze
    ws.Range( _
        ws.Cells(MAZE_TOP, MAZE_LEFT), _
        ws.Cells( _
            MAZE_TOP + MAX_MAZE_SIZE - 1, _
            MAZE_LEFT + MAX_MAZE_SIZE - 1) _
    ).Clear

    'Restore the stadium background over the cleared area so
    'no white gaps are left around a smaller maze.
    For bgRow = MAZE_TOP To MAZE_TOP + MAX_MAZE_SIZE - 1

        If ((bgRow - 1) \ 4) Mod 2 = 0 Then

            ws.Range( _
                ws.Cells(bgRow, MAZE_LEFT), _
                ws.Cells(bgRow, MAZE_LEFT + MAX_MAZE_SIZE - 1) _
            ).Interior.Color = RGB(27, 84, 50)

        Else

            ws.Range( _
                ws.Cells(bgRow, MAZE_LEFT), _
                ws.Cells(bgRow, MAZE_LEFT + MAX_MAZE_SIZE - 1) _
            ).Interior.Color = RGB(16, 53, 32)

        End If

    Next bgRow

    GeneratePerfectMaze

    If FindSolutionPath() = False Then
        Exit Function
    End If

    If CreateGuaranteedAlternativeRoute() = False Then
        Exit Function
    End If

    AddExtraRoutes

    'Find safe route again after extra routes
    If FindSolutionPath() = False Then
        Exit Function
    End If

    PlaceTraps

    PlaceGoals

    'Place World Cup Trophy at exit
    mazeData( _
        currentMazeSize - 2, _
        currentMazeSize - 2) = "E"

    playerRow = MAZE_TOP + 1
    playerCol = MAZE_LEFT + 1

    startRow = playerRow
    startCol = playerCol

    DrawFullMaze

    BuildMaze = True

End Function

'=========================================================
' GENERATE PERFECT MAZE
'=========================================================

Private Sub GeneratePerfectMaze()

    Dim visited(0 To 22, 0 To 22) As Boolean

    Dim stackR(0 To 528) As Long
    Dim stackC(0 To 528) As Long

    Dim possibleR(1 To 4) As Long
    Dim possibleC(1 To 4) As Long

    Dim currentR As Long
    Dim currentC As Long

    Dim newR As Long
    Dim newC As Long

    Dim wallR As Long
    Dim wallC As Long

    Dim optionCount As Long
    Dim selectedOption As Long

    Dim top As Long

    Dim r As Long
    Dim c As Long

    'Start with a completely closed maze
    For r = 0 To MAX_MAZE_SIZE - 1

        For c = 0 To MAX_MAZE_SIZE - 1

            mazeData(r, c) = "#"
            solutionPath(r, c) = False
            visited(r, c) = False

        Next c

    Next r

    'Starting position
    currentR = 1
    currentC = 1

    mazeData(currentR, currentC) = "."

    visited(currentR, currentC) = True

    top = 0

    stackR(top) = currentR
    stackC(top) = currentC

    Do While top >= 0

        currentR = stackR(top)
        currentC = stackC(top)

        optionCount = 0

        '---------------------------------------------
        ' UP
        '---------------------------------------------
        newR = currentR - 2
        newC = currentC

        If newR > 0 Then

            If visited(newR, newC) = False Then

                optionCount = optionCount + 1
                possibleR(optionCount) = newR
                possibleC(optionCount) = newC

            End If

        End If

        '---------------------------------------------
        ' DOWN
        '---------------------------------------------
        newR = currentR + 2
        newC = currentC

        If newR < currentMazeSize - 1 Then

            If visited(newR, newC) = False Then

                optionCount = optionCount + 1
                possibleR(optionCount) = newR
                possibleC(optionCount) = newC

            End If

        End If

        '---------------------------------------------
        ' LEFT
        '---------------------------------------------
        newR = currentR
        newC = currentC - 2

        If newC > 0 Then

            If visited(newR, newC) = False Then

                optionCount = optionCount + 1
                possibleR(optionCount) = newR
                possibleC(optionCount) = newC

            End If

        End If

        '---------------------------------------------
        ' RIGHT
        '---------------------------------------------
        newR = currentR
        newC = currentC + 2

        If newC < currentMazeSize - 1 Then

            If visited(newR, newC) = False Then

                optionCount = optionCount + 1
                possibleR(optionCount) = newR
                possibleC(optionCount) = newC

            End If

        End If

        '---------------------------------------------
        ' MOVE TO RANDOM UNVISITED CELL
        '---------------------------------------------
        If optionCount > 0 Then

            selectedOption = Int(Rnd * optionCount) + 1

            newR = possibleR(selectedOption)
            newC = possibleC(selectedOption)

            wallR = (currentR + newR) \ 2
            wallC = (currentC + newC) \ 2

            mazeData(wallR, wallC) = "."
            mazeData(newR, newC) = "."

            visited(newR, newC) = True

            top = top + 1

            stackR(top) = newR
            stackC(top) = newC

        Else

            top = top - 1

        End If

    Loop

End Sub

'=========================================================
' FIND SAFE SOLUTION PATH
'=========================================================

Private Function FindSolutionPath() As Boolean

    Dim visited(0 To 22, 0 To 22) As Boolean

    Dim previousR(0 To 22, 0 To 22) As Long
    Dim previousC(0 To 22, 0 To 22) As Long

    Dim queueR(0 To 528) As Long
    Dim queueC(0 To 528) As Long

    Dim head As Long
    Dim tail As Long

    Dim currentR As Long
    Dim currentC As Long

    Dim newR As Long
    Dim newC As Long

    Dim exitR As Long
    Dim exitC As Long

    Dim tempR As Long
    Dim tempC As Long

    Dim r As Long
    Dim c As Long

    exitR = currentMazeSize - 2
    exitC = currentMazeSize - 2

    'Reset search information
    For r = 0 To currentMazeSize - 1

        For c = 0 To currentMazeSize - 1

            visited(r, c) = False
            solutionPath(r, c) = False
            previousR(r, c) = -1
            previousC(r, c) = -1

        Next c

    Next r

    head = 0
    tail = 0

    queueR(0) = 1
    queueC(0) = 1

    visited(1, 1) = True

    Do While head <= tail

        currentR = queueR(head)
        currentC = queueC(head)

        head = head + 1

        If currentR = exitR And currentC = exitC Then
            Exit Do
        End If

        'UP
        newR = currentR - 1
        newC = currentC

        AddToSearchQueue _
            newR, newC, _
            currentR, currentC, _
            visited, previousR, previousC, _
            queueR, queueC, tail

        'DOWN
        newR = currentR + 1
        newC = currentC

        AddToSearchQueue _
            newR, newC, _
            currentR, currentC, _
            visited, previousR, previousC, _
            queueR, queueC, tail

        'LEFT
        newR = currentR
        newC = currentC - 1

        AddToSearchQueue _
            newR, newC, _
            currentR, currentC, _
            visited, previousR, previousC, _
            queueR, queueC, tail

        'RIGHT
        newR = currentR
        newC = currentC + 1

        AddToSearchQueue _
            newR, newC, _
            currentR, currentC, _
            visited, previousR, previousC, _
            queueR, queueC, tail

    Loop

    If visited(exitR, exitC) = False Then

        FindSolutionPath = False

        Exit Function

    End If

    'Trace path backwards
    currentR = exitR
    currentC = exitC

    Do

        solutionPath(currentR, currentC) = True

        If currentR = 1 And currentC = 1 Then
            Exit Do
        End If

        tempR = previousR(currentR, currentC)
        tempC = previousC(currentR, currentC)

        currentR = tempR
        currentC = tempC

    Loop

    FindSolutionPath = True

End Function

'=========================================================
' ADD SEARCH POSITION
'=========================================================

Private Sub AddToSearchQueue( _
    ByVal newR As Long, _
    ByVal newC As Long, _
    ByVal oldR As Long, _
    ByVal oldC As Long, _
    ByRef visited() As Boolean, _
    ByRef previousR() As Long, _
    ByRef previousC() As Long, _
    ByRef queueR() As Long, _
    ByRef queueC() As Long, _
    ByRef tail As Long)

    If newR < 0 Then Exit Sub

    If newR >= currentMazeSize Then Exit Sub

    If newC < 0 Then Exit Sub

    If newC >= currentMazeSize Then Exit Sub

    If visited(newR, newC) = True Then Exit Sub

    If mazeData(newR, newC) = "#" Then Exit Sub

    visited(newR, newC) = True

    previousR(newR, newC) = oldR
    previousC(newR, newC) = oldC

    tail = tail + 1

    queueR(tail) = newR
    queueC(tail) = newC

End Sub

'=========================================================
' GUARANTEED ALTERNATIVE ROUTE
'=========================================================

Private Function CreateGuaranteedAlternativeRoute() As Boolean

    Dim r As Long
    Dim c As Long

    CreateGuaranteedAlternativeRoute = False

    'Try vertical opening
    For r = 2 To currentMazeSize - 3

        For c = 1 To currentMazeSize - 2

            If mazeData(r, c) = "#" Then

                If solutionPath(r - 1, c) = True _
                    And solutionPath(r + 1, c) = True Then

                    mazeData(r, c) = "."

                    CreateGuaranteedAlternativeRoute = True

                    Exit Function

                End If

            End If

        Next c

    Next r

    'Try horizontal opening
    For r = 1 To currentMazeSize - 2

        For c = 2 To currentMazeSize - 3

            If mazeData(r, c) = "#" Then

                If solutionPath(r, c - 1) = True _
                    And solutionPath(r, c + 1) = True Then

                    mazeData(r, c) = "."

                    CreateGuaranteedAlternativeRoute = True

                    Exit Function

                End If

            End If

        Next c

    Next r

End Function

'=========================================================
' ADD EXTRA ROUTES
'=========================================================

Private Sub AddExtraRoutes()

    Dim candidateR(1 To 528) As Long
    Dim candidateC(1 To 528) As Long

    Dim candidateCount As Long
    Dim desiredRoutes As Long

    Dim r As Long
    Dim c As Long

    Dim selected As Long
    Dim i As Long

    desiredRoutes = ExtraRouteCount()

    candidateCount = 0

    For r = 1 To currentMazeSize - 2

        For c = 1 To currentMazeSize - 2

            If mazeData(r, c) = "#" Then

                If mazeData(r - 1, c) = "." _
                    And mazeData(r + 1, c) = "." Then

                    candidateCount = candidateCount + 1

                    candidateR(candidateCount) = r
                    candidateC(candidateCount) = c

                ElseIf mazeData(r, c - 1) = "." _
                    And mazeData(r, c + 1) = "." Then

                    candidateCount = candidateCount + 1

                    candidateR(candidateCount) = r
                    candidateC(candidateCount) = c

                End If

            End If

        Next c

    Next r

    For i = 1 To desiredRoutes

        If candidateCount <= 0 Then Exit For

        selected = Int(Rnd * candidateCount) + 1

        mazeData( _
            candidateR(selected), _
            candidateC(selected)) = "."

        candidateR(selected) = candidateR(candidateCount)
        candidateC(selected) = candidateC(candidateCount)

        candidateCount = candidateCount - 1

    Next i

End Sub

'=========================================================
' PLACE TRAPS
'=========================================================

Private Sub PlaceTraps()

    Dim candidateR(1 To 528) As Long
    Dim candidateC(1 To 528) As Long

    Dim candidateCount As Long
    Dim desiredTraps As Long

    Dim r As Long
    Dim c As Long

    Dim selected As Long
    Dim i As Long

    desiredTraps = TrapCountForLevel()

    candidateCount = 0

    For r = 1 To currentMazeSize - 2

        For c = 1 To currentMazeSize - 2

            If mazeData(r, c) = "." Then

                'Never place traps on guaranteed safe route
                If solutionPath(r, c) = False Then

                    'Keep traps away from start
                    If Abs(r - 1) + Abs(c - 1) > 3 Then

                        'Keep traps away from trophy
                        If Abs(r - (currentMazeSize - 2)) _
                            + Abs(c - (currentMazeSize - 2)) > 3 Then

                            candidateCount = candidateCount + 1

                            candidateR(candidateCount) = r
                            candidateC(candidateCount) = c

                        End If

                    End If

                End If

            End If

        Next c

    Next r

    For i = 1 To desiredTraps

        If candidateCount <= 0 Then Exit For

        selected = Int(Rnd * candidateCount) + 1

        mazeData( _
            candidateR(selected), _
            candidateC(selected)) = "X"

        candidateR(selected) = candidateR(candidateCount)
        candidateC(selected) = candidateC(candidateCount)

        candidateCount = candidateCount - 1

    Next i

End Sub

    Dim r As Long
    Dim c As Long

                        candidateR(candidateCount) = r
                        candidateC(candidateCount) = c

                    End If

                End If

            End If

        Next c

    Next r

    For i = 1 To desiredGoals

        If candidateCount <= 0 Then Exit For

        selected = Int(Rnd * candidateCount) + 1

        mazeData( _
            candidateR(selected), _
            candidateC(selected)) = "G"

        candidateR(selected) = candidateR(candidateCount)
        candidateC(selected) = candidateC(candidateCount)

        candidateCount = candidateCount - 1

    Next i

End Sub

'=========================================================
' MOVE PLAYER
'=========================================================

Private Sub MovePlayer( _
    ByVal rowChange As Long, _
    ByVal columnChange As Long)

    Dim newRow As Long
    Dim newCol As Long

    Dim oldGridRow As Long
    Dim oldGridCol As Long

    Dim newGridRow As Long
    Dim newGridCol As Long

    Dim target As String

    If gameRunning = False Then Exit Sub

    If gamePaused = True Then Exit Sub

    newRow = playerRow + rowChange
    newCol = playerCol + columnChange

    newGridRow = newRow - MAZE_TOP
    newGridCol = newCol - MAZE_LEFT

    If newGridRow < 0 Then Exit Sub

    If newGridRow >= currentMazeSize Then Exit Sub

    If newGridCol < 0 Then Exit Sub

    If newGridCol >= currentMazeSize Then Exit Sub

    target = mazeData(newGridRow, newGridCol)

    'Speed optimisation: redraw the worksheet only once per move.
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False

    '=========================================================
    ' DEFENDER / WALL
    '=========================================================
    If target = "#" Then

        lives = lives - 1

        score = score - LifeLossPoints()

        If score < 0 Then score = 0

        UpdateDashboard

        Application.Calculation = xlCalculationAutomatic
        Application.EnableEvents = True
        Application.ScreenUpdating = True

        PlayWallSound

        FlashLivesBar

        SetStatus _
            "DEFENDER STOPPED YOU!  LIVES: " & lives

        If lives <= 0 Then

            GameOver

        End If

        Exit Sub

    End If

    '=========================================================
    ' REMOVE OLD PLAYER
    '=========================================================
    oldGridRow = playerRow - MAZE_TOP
    oldGridCol = playerCol - MAZE_LEFT

    DrawMazeCell _
        oldGridRow, _
        oldGridCol, _
        mazeData(oldGridRow, oldGridCol)

    '=========================================================
    ' MOVE
    '=========================================================
    playerRow = newRow
    playerCol = newCol

    '=========================================================
    ' GOAL
    '=========================================================
    If target = "G" Then

        score = score + GoalValue()

        goalsCollected = goalsCollected + 1

        mazeData(newGridRow, newGridCol) = "."

    End If

    '=========================================================
    ' TRAP
    '=========================================================
    If target = "X" Then

        lives = lives - 1

        score = score - LifeLossPoints()

        If score < 0 Then score = 0

        mazeData(newGridRow, newGridCol) = "."

    End If

    '=========================================================
    ' WORLD CUP TROPHY
    '=========================================================
    If target = "E" Then

        DrawPlayer

        UpdateDashboard

        Application.Calculation = xlCalculationAutomatic
        Application.EnableEvents = True
        Application.ScreenUpdating = True

        WinLevel

        Exit Sub

    End If

    DrawPlayer

    'SPEED: the dashboard is only refreshed when the score
    'or lives actually changed - plain moves stay instant.
    If target = "G" Or target = "X" Then

        UpdateDashboard

    End If

    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    If target = "G" Then

        SetStatus _
            "GOAL!  +" & GoalValue() & " POINTS"

        PlayGoalSound

    End If

    If target = "X" Then

        PlayTrapSound

        FlashLivesBar

        SetStatus _
            "TRAP!  LIVES REMAINING: " & lives

    End If

    If lives <= 0 Then

        GameOver

    End If

End Sub

    Dim bgRow As Long

    ws.Cells.Interior.Color = RGB(21, 68, 41)

'=========================================================
' TIMER
'=========================================================

Private Function LevelTimeRemaining() As Long

    Dim elapsedLevelSeconds As Long

    elapsedLevelSeconds = GetElapsedSeconds() - levelStartElapsedSeconds

    LevelTimeRemaining = LEVEL_TIME_LIMIT - elapsedLevelSeconds

    If LevelTimeRemaining < 0 Then
        LevelTimeRemaining = 0
    End If

End Function


Public Sub UpdateTimer()

    Dim secondsPassed As Long

    nextTickScheduled = False

    If gameRunning = False Then Exit Sub

    If gamePaused = True Then Exit Sub

    secondsPassed = GetElapsedSeconds()

    'Speed optimisation: write the timer cells directly
    'without toggling the screen repaint.
    ThisWorkbook.Worksheets(SHEET_NAME) _
        .Range("AB10").value = _
        FormatGameTime(secondsPassed)

    ThisWorkbook.Worksheets(SHEET_NAME) _
        .Range("AB12").value = _
        FormatGameTime(LevelTimeRemaining())

    If LevelTimeRemaining() <= 0 Then

        SetStatus "TIME UP!"

        GameOver

        Exit Sub

    End If

    nextTick = Now + TimeSerial(0, 0, 1)

    Application.OnTime _
        EarliestTime:=nextTick, _
        Procedure:=TimerProcedureName(), _
        Schedule:=True

    nextTickScheduled = True

End Sub

'=========================================================
' ELAPSED TIME
'=========================================================

Private Function GetElapsedSeconds() As Long

    Dim secondsPassed As Double

    If gamePaused = True Then

        secondsPassed = _
            (pauseStarted - gameStart) * 86400 _
            - totalPausedSeconds

    Else

        secondsPassed = _
            (Now - gameStart) * 86400 _
            - totalPausedSeconds

    End If

    If secondsPassed < 0 Then

        secondsPassed = 0

    End If

    GetElapsedSeconds = CLng(secondsPassed)

End Function

'=========================================================
' FORMAT TIME
'=========================================================

Private Function FormatGameTime( _
    ByVal totalSeconds As Long) As String

    Dim minutes As Long
    Dim seconds As Long

    minutes = totalSeconds \ 60

    seconds = totalSeconds Mod 60

    FormatGameTime = _
        Format(minutes, "00") & ":" & _
        Format(seconds, "00")

End Function

'=========================================================
' MUSIC ENGINE - CHILLED STADIUM THEME
'
' Three relaxed, football-themed songs are generated as
' WAV files and played through Windows MCI:
'
'   Track 1 "Stadium Chill"     - the original soft groove.
'   Track 2 "Samba Breeze"      - bright bossa with
'                                 syncopated off-beat bass
'                                 and marimba-style plucks.
'   Track 3 "Champion's Anthem" - upbeat but relaxed
'                                 stadium-anthem groove in
'                                 C major with a bold,
'                                 rising melody.
'
' SHUFFLE: when a track finishes, a random DIFFERENT
' track starts automatically, so the music stays fresh.
'
' Press M during play, or run the ToggleMusic macro, to
' switch the music on or off.
'=========================================================

Public Sub StartMusic()

    If musicOn = False Then Exit Sub

    InitAudio

    If audioReady = False Then Exit Sub

    If currentMusicTrack < 1 Or currentMusicTrack > 3 Then
        currentMusicTrack = 1
    End If

    If musicPlaying = False Then

        PlayMusicTrack currentMusicTrack

    End If

    StartMusicTimer

End Sub


Public Sub StopMusic()

    Dim i As Long

    musicPlaying = False

    StopMusicTimer

    For i = 1 To 3

        MciExec "stop WCMusic" & CStr(i)

    Next i

End Sub


'Run this macro (or press M during play) to switch the
'music on / off.
Public Sub ToggleMusic()

    musicOn = Not musicOn

    If musicOn Then

        If gamePaused = False Then StartMusic

        If gameRunning = True And gamePaused = False Then
            SetStatus "MUSIC ON"
        End If

    Else

        StopMusic

        If gameRunning = True And gamePaused = False Then
            SetStatus "MUSIC OFF"
        End If

    End If

End Sub

Private Sub KeepMusicLoop()

    'SHUFFLE: when the current track has finished, a random
    'DIFFERENT track starts, so any of the three songs can
    'play next and none repeats back to back.

    Dim modeText As String
    Dim offset As Long
    Dim newTrack As Long

    If musicBusy Then Exit Sub
    If musicPlaying = False Then Exit Sub

    musicBusy = True

    modeText = MciQuery("status WCMusic" & CStr(currentMusicTrack) & " mode")

    If Len(modeText) = 0 _
        Or InStr(1, modeText, "playing", vbTextCompare) = 0 Then

        'Pick randomly from the other two tracks.
        offset = Int(Rnd * 2)   '0 or 1

        newTrack = currentMusicTrack + offset + 1

        If newTrack > 3 Then newTrack = newTrack - 3

        currentMusicTrack = newTrack

        PlayMusicTrack currentMusicTrack

    End If

    musicBusy = False

End Sub

'=========================================================
' AUDIO GENERATION AND PLAYBACK (MCI)
'
' Music and sound effects are synthesised into small WAV
' files in the temp folder (cached, so they are only
' generated once) and played through Windows MCI.
'=========================================================

Private Sub MciExec(ByVal commandText As String)

    On Error Resume Next

    mciSendStringW StrPtr(commandText), 0, 0, 0

    On Error GoTo 0

End Sub


Private Function MciQuery(ByVal commandText As String) As String

    Dim buf As String

    buf = String$(260, Chr$(0))

    On Error Resume Next

    mciSendStringW StrPtr(commandText), StrPtr(buf), 260, 0

    On Error GoTo 0

    MciQuery = Replace(buf, Chr$(0), "")

End Function


Private Function MusicTrackFile(ByVal trackNo As Long) As String

    'Track file names are versioned so that updated tunes
    'are regenerated instead of replaying an old cached
    'file with the same name.

    Dim suffix As String

    suffix = ""

    If trackNo = 3 Then suffix = "_v2"

    MusicTrackFile = audioFolder & "\WCEscape_Track" & CStr(trackNo) & suffix & ".wav"

End Function


Private Sub InitAudio()

    Dim statusText As String
    Dim i As Long

    If audioReady Then Exit Sub

    audioFolder = Environ("TEMP")
    If audioFolder = "" Then audioFolder = Environ("TMP")
    If audioFolder = "" Then audioFolder = CurDir$()

    'Close any aliases left over from an earlier session.
    On Error Resume Next

    For i = 1 To 3
        MciExec "close WCMusic" & CStr(i)
    Next i

    MciExec "close WCGoal"
    MciExec "close WCTrap"
    MciExec "close WCWall"
    MciExec "close WCWin"
    MciExec "close WCOver"
    MciExec "close WCClick"

    On Error GoTo 0

    'Music tracks: regenerate any that are missing. The
    'file names are versioned, so updated tunes are picked
    'up automatically on the first run.
    For i = 1 To 3

        If Len(Dir(MusicTrackFile(i))) = 0 Then
            GenerateMusicFile i
        End If

    Next i

    'Sound effects: only generated if missing (cached).
    If Len(Dir(audioFolder & "\WCEscape_Goal.wav")) = 0 _
        Or Len(Dir(audioFolder & "\WCEscape_Trap.wav")) = 0 _
        Or Len(Dir(audioFolder & "\WCEscape_Wall.wav")) = 0 _
        Or Len(Dir(audioFolder & "\WCEscape_Win.wav")) = 0 _
        Or Len(Dir(audioFolder & "\WCEscape_Over.wav")) = 0 _
        Or Len(Dir(audioFolder & "\WCEscape_Click.wav")) = 0 Then

        GenerateSfxFiles

    End If

    On Error Resume Next

    For i = 1 To 3

        MciExec "open """ & MusicTrackFile(i) & """ alias WCMusic" & CStr(i)

        'Keep the music soft under the sound effects.
        MciExec "set WCMusic" & CStr(i) & " volume to 600"

    Next i

    MciExec "open """ & audioFolder & "\WCEscape_Goal.wav"" alias WCGoal"
    MciExec "open """ & audioFolder & "\WCEscape_Trap.wav"" alias WCTrap"
    MciExec "open """ & audioFolder & "\WCEscape_Wall.wav"" alias WCWall"
    MciExec "open """ & audioFolder & "\WCEscape_Win.wav"" alias WCWin"
    MciExec "open """ & audioFolder & "\WCEscape_Over.wav"" alias WCOver"
    MciExec "open """ & audioFolder & "\WCEscape_Click.wav"" alias WCClick"

    On Error GoTo 0

    statusText = MciQuery("status WCMusic1 mode")

    If Len(statusText) > 0 Then audioReady = True

End Sub

Private Sub GenerateMusicFile(ByVal trackNo As Long)

    Dim sampleRate As Long
    Dim stepSec As Double
    Dim stepsPerPass As Long
    Dim passes As Long
    Dim nSamples As Long
    Dim samples() As Double
    Dim i As Long
    Dim t As Double
    Dim stepIdx As Long
    Dim patternStep As Long
    Dim chordIdx As Long
    Dim bassIdx As Long
    Dim stepOf8 As Long
    Dim chordRow As Variant
    Dim chord1 As Double
    Dim chord2 As Double
    Dim chord3 As Double
    Dim bassNote As Long
    Dim chordLen As Double
    Dim chordPos As Double
    Dim env As Double
    Dim v As Double
    Dim melodyN As Long
    Dim noteFreq As Double
    Dim noteStart As Double
    Dim notePos As Double
    Dim bassFreq As Double
    Dim bassPos As Double
    Dim crowdAvg As Double
    Dim pi2 As Double

    sampleRate = 11025
    stepsPerPass = 32
    passes = 2
    pi2 = 8 * Atn(1)

    LoadMusicTune trackNo

    stepSec = musicStepSec

    nSamples = CLng(stepSec * stepsPerPass * passes * sampleRate)
    ReDim samples(0 To nSamples - 1)

    chordLen = stepSec * 8
    crowdAvg = 0

    For i = 0 To nSamples - 1

        t = i / sampleRate

        stepIdx = Int(t / stepSec)
        patternStep = stepIdx Mod stepsPerPass
        chordIdx = Int(patternStep / 8)
        bassIdx = Int(patternStep / 4)
        stepOf8 = (patternStep Mod 8) + 1

        chordRow = musicChords(chordIdx)
        chord1 = chordRow(0)
        chord2 = chordRow(1)
        chord3 = chordRow(2)

        bassNote = musicBass(bassIdx)

        chordPos = t - Int(t / chordLen) * chordLen

        env = chordPos / musicPadAttack
        If env > 1 Then env = 1
        If (chordLen - chordPos) / 0.35 < env Then env = (chordLen - chordPos) / 0.35
        If env < 0 Then env = 0

        'Soft chord pads.
        v = env * 0.05 * (Sin(pi2 * chord1 * t) _
            + Sin(pi2 * chord2 * t) + Sin(pi2 * chord3 * t))

        'Melody - the timbre differs per track (pluck,
        'marimba, smooth lead or bell).
        melodyN = musicMelody(patternStep)

        If melodyN > 0 Then

            noteStart = stepIdx * stepSec
            notePos = t - noteStart

            If notePos < musicNoteLen Then

                noteFreq = 440# * 2 ^ ((melodyN - 69) / 12)

                env = notePos / 0.015
                If env > 1 Then env = 1
                env = env * Exp(-notePos * musicNoteDecay)

                v = v + env * 0.1 * (Sin(pi2 * noteFreq * t) _
                    + musicTone2Amp * Sin(pi2 * noteFreq * musicTone2Ratio * t))

            End If

        End If

        'Bass - the rhythm pattern differs per track.
        If Mid$(musicBassSteps, stepOf8, 1) = "1" Then

            bassPos = t - stepIdx * stepSec

            If bassPos < 1.2 Then

                bassFreq = 440# * 2 ^ ((bassNote - 69) / 12)

                v = v + Exp(-bassPos * musicBassDecay) * 0.11 * Sin(pi2 * bassFreq * t)

            End If

        End If

        'Kick - pattern differs per track.
        If Mid$(musicKickSteps, stepOf8, 1) = "1" Then

            bassPos = t - stepIdx * stepSec

            If bassPos < 0.1 Then
                v = v + Exp(-bassPos * 45) * 0.1 * Sin(pi2 * 60 * t)
            End If

        End If

        'Shaker - pattern differs per track.
        If Mid$(musicShakerSteps, stepOf8, 1) = "1" Then

            bassPos = t - stepIdx * stepSec

            If bassPos < 0.04 Then
                v = v + (Rnd - 0.5) * 0.04 * Exp(-bassPos * 90)
            End If

        End If

        'Low crowd ambience rumble.
        crowdAvg = crowdAvg * 0.996 + (Rnd * 2 - 1) * 0.004
        v = v + crowdAvg * musicCrowdAmp * (0.75 + 0.25 * Sin(pi2 * t / 9))

        'Soft limiting.
        If v > 0.9 Then v = 0.9
        If v < -0.9 Then v = -0.9

        samples(i) = v

    Next i

    WriteWavFile samples, sampleRate, MusicTrackFile(trackNo)

End Sub

Private Sub SfxBegin(ByVal durationSec As Double)

    Dim n As Long

    sfxRate = 11025

    n = CLng(durationSec * sfxRate)
    If n < 2 Then n = 2

    ReDim sfxBuf(0 To n - 1)

End Sub


Private Sub SfxTone( _
    ByVal startSec As Double, _
    ByVal durSec As Double, _
    ByVal freq As Double, _
    ByVal amp As Double, _
    ByVal harmonic2 As Double, _
    ByVal decayRate As Double)

    Dim i As Long
    Dim i0 As Long
    Dim i1 As Long
    Dim ph As Double
    Dim env As Double
    Dim pi2 As Double

    pi2 = 8 * Atn(1)

    i0 = CLng(startSec * sfxRate)
    i1 = i0 + CLng(durSec * sfxRate) - 1

    If i0 < 0 Then i0 = 0
    If i1 > UBound(sfxBuf) Then i1 = UBound(sfxBuf)
    If i1 < i0 Then Exit Sub

    For i = i0 To i1

        ph = (i - i0) / sfxRate

        env = Exp(-ph * decayRate)

        If ph < 0.006 Then
            env = env * (ph / 0.006)
        End If

        sfxBuf(i) = sfxBuf(i) + env * amp * _
            (Sin(pi2 * freq * ph) + harmonic2 * Sin(pi2 * freq * 2 * ph))

    Next i

End Sub


Private Sub SfxNoise( _
    ByVal startSec As Double, _
    ByVal durSec As Double, _
    ByVal amp As Double, _
    ByVal decayRate As Double)

    Dim i As Long
    Dim i0 As Long
    Dim i1 As Long
    Dim ph As Double
    Dim env As Double

    i0 = CLng(startSec * sfxRate)
    i1 = i0 + CLng(durSec * sfxRate) - 1

    If i0 < 0 Then i0 = 0
    If i1 > UBound(sfxBuf) Then i1 = UBound(sfxBuf)
    If i1 < i0 Then Exit Sub

    For i = i0 To i1

        ph = (i - i0) / sfxRate

        env = Exp(-ph * decayRate)

        sfxBuf(i) = sfxBuf(i) + (Rnd - 0.5) * 2 * amp * env

    Next i

End Sub


Private Sub WriteWavFile( _
    ByRef samples() As Double, _
    ByVal sampleRate As Long, _
    ByVal filePath As String)

    Dim hdr(0 To 43) As Byte
    Dim wavBytes() As Byte
    Dim nSamples As Long
    Dim dataSize As Long
    Dim riffSize As Long
    Dim i As Long
    Dim pcm As Long
    Dim u As Long
    Dim fnum As Integer

    nSamples = UBound(samples) + 1
    dataSize = nSamples * 2
    riffSize = 36 + dataSize

    'RIFF header (16-bit mono PCM).
    hdr(0) = 82: hdr(1) = 73: hdr(2) = 70: hdr(3) = 70
    PutLongInBytes hdr, 4, riffSize
    hdr(8) = 87: hdr(9) = 65: hdr(10) = 86: hdr(11) = 69
    hdr(12) = 102: hdr(13) = 109: hdr(14) = 116: hdr(15) = 32
    PutLongInBytes hdr, 16, 16
    hdr(20) = 1: hdr(21) = 0
    hdr(22) = 1: hdr(23) = 0
    PutLongInBytes hdr, 24, sampleRate
    PutLongInBytes hdr, 28, sampleRate * 2
    hdr(32) = 2: hdr(33) = 0
    hdr(34) = 16: hdr(35) = 0
    hdr(36) = 100: hdr(37) = 97: hdr(38) = 116: hdr(39) = 97
    PutLongInBytes hdr, 40, dataSize

    ReDim wavBytes(0 To dataSize - 1)

    For i = 0 To nSamples - 1

        pcm = CLng(samples(i) * 32000)

        If pcm > 32767 Then pcm = 32767
        If pcm < -32768 Then pcm = -32768

        u = pcm And &HFFFF&

        wavBytes(i * 2) = CByte(u And 255)
        wavBytes(i * 2 + 1) = CByte(u \ 256)

    Next i

    On Error Resume Next

    Kill filePath

    On Error GoTo 0

    fnum = FreeFile

    Open filePath For Binary Access Write As #fnum
    Put #fnum, , hdr
    Put #fnum, , wavBytes
    Close #fnum

End Sub


Private Sub PutLongInBytes(ByRef b() As Byte, ByVal offset As Long, ByVal value As Long)

    b(offset) = value And 255
    b(offset + 1) = (value \ 256) And 255
    b(offset + 2) = (value \ 65536) And 255
    b(offset + 3) = (value \ 16777216) And 255

End Sub

Public Sub PauseGame()

    If gameRunning = False Then

        MsgBox "Start the game first.", _
            vbInformation, _
            "World Cup Escape"

        Exit Sub

    End If

    If gamePaused = True Then

        MsgBox "The game is already paused.", _
            vbInformation, _
            "World Cup Escape"

        Exit Sub

    End If

    pauseStarted = Now

    gamePaused = True

    CancelTimer

    StopMusic

    SetStatus "PAUSED"

End Sub

'=========================================================
' RESUME
'=========================================================

Public Sub ResumeGame()

    If gameRunning = False Then

        MsgBox "Start the game first.", _
            vbInformation, _
            "World Cup Escape"

        Exit Sub

    End If

    If gamePaused = False Then

        MsgBox "The game is not paused.", _
            vbInformation, _
            "World Cup Escape"

        Exit Sub

    End If

    totalPausedSeconds = _
        totalPausedSeconds + _
        ((Now - pauseStarted) * 86400)

    gamePaused = False

    StartMusic

    SetStatus "PLAYING"

    UpdateTimer

End Sub

' WORLD CUP TROPHY ANIMATION
'
' A short celebration shown on the pitch when the player
' beats level 5: a golden World Cup style trophy with the
' recognisable silhouette - the gold globe with continents
' held aloft, a small figure's head just beneath it, a
' body that rises narrow from the base, swells to its
' widest in the lower-middle, then sweeps up into two
' spiral arms whose tips curl outward and hug the globe's
' lower sides - all on the malachite base (green band,
' engraved gold "FIFA WORLD CUP" band, green band).
' The whole trophy gently pulses while a ring of stars
' orbits it, followed by the champion message box.
'=========================================================

Private Sub PlayTrophyAnimation()

    Dim ws As Worksheet
    Dim bannerShp As Shape
    Dim stars(0 To 7) As Shape
    Dim partShp As Shape
    Dim bodyShp As Shape
    Dim partNames(0 To 24) As String
    Dim partL(0 To 24) As Double
    Dim partT(0 To 24) As Double
    Dim partW(0 To 24) As Double
    Dim partH(0 To 24) As Double
    Dim partCount As Long
    Dim cx As Double
    Dim cy As Double
    Dim tcy As Double
    Dim i As Long
    Dim j As Long
    Dim ang As Double
    Dim rad As Double
    Dim pulse As Double
    Dim animStart As Double
    Dim pi As Double

    Dim bodyPts As Variant
    Dim ff As FreeformBuilder

    On Error Resume Next

    Set ws = ThisWorkbook.Worksheets(SHEET_NAME)

    pi = Atn(1) * 4

    'Remove any leftover animation shapes first.
    DeleteTrophyShapes ws

    Application.ScreenUpdating = True

    'Centre of the current maze.
    cx = ws.Cells(MAZE_TOP + (currentMazeSize \ 2), _
        MAZE_LEFT + (currentMazeSize \ 2)).Left _
        + ws.Cells(MAZE_TOP + (currentMazeSize \ 2), _
        MAZE_LEFT + (currentMazeSize \ 2)).Width / 2

    cy = ws.Cells(MAZE_TOP + (currentMazeSize \ 2), _
        MAZE_LEFT + (currentMazeSize \ 2)).top _
        + ws.Cells(MAZE_TOP + (currentMazeSize \ 2), _
        MAZE_LEFT + (currentMazeSize \ 2)).Height / 2

    tcy = cy - 10

    '-----------------------------------------------------
    ' MALACHITE BASE - green band, gold engraved band,
    ' green band (bottom to top of the real trophy base).
    '-----------------------------------------------------

    'Lower green malachite band (widest).
    Set partShp = ws.Shapes.AddShape(msoShapeRoundedRectangle, _
        cx - 48, tcy + 130, 96, 18)
    partShp.Name = "WCTrophyAnim__BASEBOT"
    partShp.Fill.ForeColor.RGB = RGB(9, 44, 30)
    partShp.Line.ForeColor.RGB = RGB(170, 120, 10)
    partShp.Line.Weight = 1.25
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    'Gold engraved band with the trophy's inscription.
    Set partShp = ws.Shapes.AddShape(msoShapeRoundedRectangle, _
        cx - 44, tcy + 112, 88, 18)
    partShp.Name = "WCTrophyAnim__BAND"
    partShp.Fill.ForeColor.RGB = RGB(255, 205, 50)
    partShp.Line.ForeColor.RGB = RGB(170, 120, 10)
    partShp.Line.Weight = 1
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    With partShp.TextFrame2
        .WordWrap = msoFalse
        .MarginLeft = 0
        .MarginRight = 0
        .MarginTop = 0
        .MarginBottom = 0
        .TextRange.Text = "FIFA WORLD CUP"
        .TextRange.Font.Size = 7
        .TextRange.Font.Bold = msoTrue
        .TextRange.Font.Fill.ForeColor.RGB = RGB(9, 44, 30)
        .TextRange.ParagraphFormat.Alignment = msoAlignCenter
        .VerticalAnchor = msoAnchorMiddle
    End With

    'Upper green malachite band (narrower).
    Set partShp = ws.Shapes.AddShape(msoShapeRoundedRectangle, _
        cx - 38, tcy + 96, 76, 16)
    partShp.Name = "WCTrophyAnim__BASETOP"
    partShp.Fill.ForeColor.RGB = RGB(12, 58, 40)
    partShp.Line.ForeColor.RGB = RGB(170, 120, 10)
    partShp.Line.Weight = 1.25
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    '-----------------------------------------------------
    ' GOLD BODY - the true World Cup silhouette: narrow at
    ' the base, swelling to its widest in the lower-middle,
    ' narrowing again, then flaring up into two spiral arms
    ' whose tips curl outward and hug the globe.
    ' (Drawn as a freeform, with a flipped-trapezoid
    ' fallback for older Excel versions.)
    '-----------------------------------------------------

    bodyPts = Array( _
        -22, 96, -30, 76, -44, 48, -52, 18, -50, -8, -45, -28, _
        -41, -44, -47, -56, -51, -66, -43, -71, -30, -74, _
        0, -78, _
        30, -74, 43, -71, 51, -66, 47, -56, 41, -44, 45, -28, _
        50, -8, 52, 18, 44, 48, 30, 76, 22, 96)

    Set bodyShp = Nothing

    On Error Resume Next

    Set ff = ws.Shapes.BuildFreeform(msoEditingCorner, _
        cx + bodyPts(0), tcy + bodyPts(1))

    For j = 2 To UBound(bodyPts) Step 2

        ff.AddNodes msoSegmentLine, msoEditingCorner, _
            cx + bodyPts(j), tcy + bodyPts(j + 1)

    Next j

    Set bodyShp = ff.ConvertToShape

    On Error GoTo 0

    If bodyShp Is Nothing Then

        'Fallback: flipped trapezoid (wide at the top).
        Set bodyShp = ws.Shapes.AddShape(msoShapeTrapezoid, _
            cx - 52, tcy - 78, 104, 174)
        bodyShp.Flip msoFlipVertical

        On Error Resume Next
        bodyShp.Adjustments.Item(1) = 0.3
        On Error GoTo 0

    End If

    bodyShp.Name = "WCTrophyAnim__BODY"
    bodyShp.Fill.ForeColor.RGB = RGB(255, 205, 50)
    bodyShp.Line.ForeColor.RGB = RGB(170, 120, 10)
    bodyShp.Line.Weight = 1.25
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, bodyShp

    'Two interlaced spiral wing streaks crossing the body.
    Set partShp = ws.Shapes.AddShape(msoShapeOval, _
        cx - 36, tcy - 36, 16, 88)
    partShp.Name = "WCTrophyAnim__SPIRALL"
    partShp.Fill.ForeColor.RGB = RGB(214, 150, 20)
    partShp.Fill.Transparency = 0.45
    partShp.Line.Visible = msoFalse
    partShp.rotation = 12
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    Set partShp = ws.Shapes.AddShape(msoShapeOval, _
        cx + 20, tcy - 36, 16, 88)
    partShp.Name = "WCTrophyAnim__SPIRALR"
    partShp.Fill.ForeColor.RGB = RGB(214, 150, 20)
    partShp.Fill.Transparency = 0.45
    partShp.Line.Visible = msoFalse
    partShp.rotation = -12
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    'Soft highlight on the left of the body.
    Set partShp = ws.Shapes.AddShape(msoShapeOval, _
        cx - 29, tcy - 40, 9, 78)
    partShp.Name = "WCTrophyAnim__BODYHL"
    partShp.Fill.ForeColor.RGB = RGB(255, 240, 180)
    partShp.Fill.Transparency = 0.55
    partShp.Line.Visible = msoFalse
    partShp.rotation = 10
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    'Soft shading on the right of the body.
    Set partShp = ws.Shapes.AddShape(msoShapeOval, _
        cx + 25, tcy - 30, 10, 76)
    partShp.Name = "WCTrophyAnim__BODYSH"
    partShp.Fill.ForeColor.RGB = RGB(205, 140, 15)
    partShp.Fill.Transparency = 0.5
    partShp.Line.Visible = msoFalse
    partShp.rotation = -8
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    '-----------------------------------------------------
    ' THE FIGURE'S HEAD just beneath the globe (its top is
    ' tucked behind the globe, as on the real trophy).
    '-----------------------------------------------------

    Set partShp = ws.Shapes.AddShape(msoShapeOval, _
        cx - 7, tcy - 40, 14, 20)
    partShp.Name = "WCTrophyAnim__HEAD"
    partShp.Fill.ForeColor.RGB = RGB(255, 225, 120)
    partShp.Line.ForeColor.RGB = RGB(170, 120, 10)
    partShp.Line.Weight = 1
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    '-----------------------------------------------------
    ' THE GLOBE with continents and a highlight, cradled
    ' by the spiral arms.
    '-----------------------------------------------------

    Set partShp = ws.Shapes.AddShape(msoShapeOval, _
        cx - 46, tcy - 124, 92, 92)
    partShp.Name = "WCTrophyAnim__GLOBE"
    partShp.Fill.ForeColor.RGB = RGB(255, 205, 50)
    partShp.Line.ForeColor.RGB = RGB(170, 120, 10)
    partShp.Line.Weight = 1.5
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    Set partShp = ws.Shapes.AddShape(msoShapeOval, _
        cx - 36, tcy - 106, 26, 18)
    partShp.Name = "WCTrophyAnim__CONT1"
    partShp.Fill.ForeColor.RGB = RGB(206, 150, 20)
    partShp.Line.Visible = msoFalse
    partShp.rotation = -25
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    Set partShp = ws.Shapes.AddShape(msoShapeOval, _
        cx + 2, tcy - 88, 24, 16)
    partShp.Name = "WCTrophyAnim__CONT2"
    partShp.Fill.ForeColor.RGB = RGB(206, 150, 20)
    partShp.Line.Visible = msoFalse
    partShp.rotation = 20
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    Set partShp = ws.Shapes.AddShape(msoShapeOval, _
        cx - 8, tcy - 56, 16, 11)
    partShp.Name = "WCTrophyAnim__CONT3"
    partShp.Fill.ForeColor.RGB = RGB(206, 150, 20)
    partShp.Line.Visible = msoFalse
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    Set partShp = ws.Shapes.AddShape(msoShapeOval, _
        cx - 34, tcy - 118, 20, 14)
    partShp.Name = "WCTrophyAnim__GLOBEHL"
    partShp.Fill.ForeColor.RGB = RGB(255, 255, 255)
    partShp.Fill.Transparency = 0.55
    partShp.Line.Visible = msoFalse
    AddTrophyPart partNames, partL, partT, partW, partH, partCount, partShp

    '-----------------------------------------------------
    ' Champion banner.
    '-----------------------------------------------------

    Set bannerShp = ws.Shapes.AddTextbox( _
        msoTextOrientationHorizontal, _
        cx - 190, tcy + 168, 380, 36)

    bannerShp.Name = "WCTrophyAnim__BANNER"
    bannerShp.ZOrder msoBringToFront

    With bannerShp
        .Fill.ForeColor.RGB = RGB(15, 35, 60)
        .Line.ForeColor.RGB = RGB(255, 215, 0)
        .Line.Weight = 1.5
    End With

    With bannerShp.TextFrame2
        .TextRange.Text = "WORLD CUP CHAMPION!"
        .TextRange.Font.Size = 18
        .TextRange.Font.Bold = msoTrue
        .TextRange.Font.Fill.ForeColor.RGB = RGB(255, 215, 0)
        .TextRange.ParagraphFormat.Alignment = msoAlignCenter
        .VerticalAnchor = msoAnchorMiddle
    End With

    '-----------------------------------------------------
    ' Ring of golden stars.
    '-----------------------------------------------------

    rad = 150

    For i = 0 To 7

        Set stars(i) = ws.Shapes.AddShape( _
            msoShape5pointStar, cx - 12, cy - 12, 24, 24)

        stars(i).Name = "WCTrophyAnim__STAR" & CStr(i)
        stars(i).Fill.ForeColor.RGB = RGB(255, 215, 0)
        stars(i).Line.Visible = msoFalse
        stars(i).ZOrder msoBringToFront

    Next i

    'Animate for about two seconds.
    animStart = Timer
    ang = 0

    Do While Timer - animStart < 2

        ang = ang + 9

        'Trophy gently pulses around its centre.
        pulse = 1 + (0.07 * Sin(ang * pi / 30))

        For i = 0 To partCount - 1

            Set partShp = ws.Shapes(partNames(i))

            partShp.Width = partW(i) * pulse
            partShp.Height = partH(i) * pulse
            partShp.Left = cx + (partL(i) - cx) * pulse
            partShp.top = tcy + (partT(i) - tcy) * pulse

        Next i

        'Stars orbit the trophy.
        For i = 0 To 7
            stars(i).Left = cx + (Cos((ang + i * 45) * pi / 180) * rad) _
                - (stars(i).Width / 2)
            stars(i).top = (tcy - 10) + _
                (Sin((ang + i * 45) * pi / 180) * rad * 0.5) _
                - (stars(i).Height / 2)
        Next i

        DoEvents

    Loop

    DeleteTrophyShapes ws

    On Error GoTo 0

End Sub


Private Sub AddTrophyPart( _
    ByRef partNames() As String, _
    ByRef partL() As Double, _
    ByRef partT() As Double, _
    ByRef partW() As Double, _
    ByRef partH() As Double, _
    ByRef partCount As Long, _
    ByVal shp As Shape)

    partNames(partCount) = shp.Name
    partL(partCount) = shp.Left
    partT(partCount) = shp.top
    partW(partCount) = shp.Width
    partH(partCount) = shp.Height
    partCount = partCount + 1

End Sub


Private Sub DeleteTrophyShapes(ByVal ws As Worksheet)

    Dim shp As Shape

    On Error Resume Next

    For Each shp In ws.Shapes

        If Left$(shp.Name, 14) = "WCTrophyAnim__" Then
            shp.Delete
        End If

    Next shp

    On Error GoTo 0

End Sub

'=========================================================
' GAME OVER
'=========================================================

Private Sub GameOver()

    Dim levelTime As Long
    Dim answer As VbMsgBoxResult
    Dim msg As String

    levelTime = GetElapsedSeconds()

    gameRunning = False

    CancelTimer

    StopMusic

    PlayGameOverTune

    SaveHighScore _
        score, _
        goalsCollected, _
        levelTime, _
        StageName()

    SetStatus "ELIMINATED"

    msg = "YOU HAVE BEEN ELIMINATED!" & vbCrLf & vbCrLf

    msg = msg & "Tournament Stage: " _
        & StageName() & vbCrLf

    msg = msg & "Goals: " _
        & goalsCollected & vbCrLf

    msg = msg & "Score: " _
        & score & vbCrLf & vbCrLf

    msg = msg & "Would you like another chance?" _
        & vbCrLf & vbCrLf

    msg = msg & _
        "Your score and goals from previous completed " _
        & "levels will be preserved."

    answer = MsgBox( _
        msg, _
        vbYesNo + vbExclamation, _
        "ELIMINATED")

    If answer = vbYes Then

        BindArrowKeys

        RetryGame

    Else

        UnbindArrowKeys

        CloseSound

    End If

End Sub

'=========================================================
' RETRY CURRENT LEVEL
'
' Restores score/goals to the beginning-of-level
' checkpoint and creates a completely new maze.
'=========================================================

Private Sub RetryGame()

    Dim oldScreenUpdating As Boolean

    oldScreenUpdating = Application.ScreenUpdating

    '=========================================================
    ' SPEED OPTIMISATION
    '
    ' Use the same loading optimisation as StartLevel.
    ' This prevents Excel from repainting the worksheet
    ' while the new maze is being generated and drawn.
    '=========================================================
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False

    On Error GoTo RestoreScreenAndRaise

    score = levelStartScore

    goalsCollected = levelStartGoals

    lives = LivesForLevel()

    gameRunning = True
    gamePaused = False

    'Bring back the chilled music for the new attempt.
    StartMusic

    currentMazeSize = MazeSizeForLevel()

    If BuildMaze() = False Then

        gameRunning = False

        Application.Calculation = xlCalculationAutomatic
        Application.EnableEvents = True
        Application.ScreenUpdating = oldScreenUpdating

        MsgBox _
            "The new maze could not be created." & vbCrLf & _
            "Please try again.", _
            vbCritical, _
            "World Cup Escape"

        Exit Sub

    End If

    levelStartElapsedSeconds = GetElapsedSeconds()

    playerRow = MAZE_TOP + 1
    playerCol = MAZE_LEFT + 1

    startRow = playerRow
    startCol = playerCol

    UpdateDashboard

    'Draw the team flag once per retry (not on every move).
    RefreshDashboardTeamFlag

    SetStatus "NEW ATTEMPT - GOOD LUCK!"

    UpdateTimer

    'Show the completed new maze only after everything
    'has finished loading.
    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    Application.ScreenUpdating = oldScreenUpdating

    Exit Sub

RestoreScreenAndRaise:

    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    Application.ScreenUpdating = oldScreenUpdating

    Err.Raise Err.Number, Err.Source, Err.Description

End Sub

'=========================================================
' RESTART CURRENT LEVEL MANUALLY
'=========================================================

Public Sub RestartCurrentLevel()

    If currentLevel < 1 Then

        StartGame

        Exit Sub

    End If

    RetryGame

End Sub

'=========================================================
' TIMER PROCEDURE NAME
'=========================================================

Private Function TimerProcedureName() As String

    TimerProcedureName = _
        "'" & Replace( _
        ThisWorkbook.Name, _
        "'", _
        "''") & "'!UpdateTimer"

End Function

'=========================================================
' CANCEL TIMER
'=========================================================

Private Sub CancelTimer()

    On Error Resume Next

    If nextTickScheduled = True Then

        Application.OnTime _
            EarliestTime:=nextTick, _
            Procedure:=TimerProcedureName(), _
            Schedule:=False

    End If

    nextTickScheduled = False

    nextTick = 0

    On Error GoTo 0

End Sub

Private Sub RefreshDashboardTeamFlag()

    Dim ws As Worksheet
    Dim teamIndex As Long
    Dim target As Range

    Set ws = ThisWorkbook.Worksheets(SHEET_NAME)

    DeleteDashboardFlagShapes ws

    If selectedTeam = "" Then Exit Sub

    teamIndex = TeamIndexFromName(selectedTeam)

    If teamIndex < 0 Then Exit Sub

    Set target = ws.Range("AB2")

Private Sub RefreshHighScoreFlags()

    Dim r As Long
    Dim teamName As String
    Dim teamIndex As Long

    Set ws = ThisWorkbook.Worksheets(SHEET_NAME)

    DeleteHighScoreFlagShapes ws

    On Error Resume Next

    For Each target In ws.Range("AI3:AI12")

        teamName = CStr(target.value)

        If teamName <> "" Then

            teamName = ExtractTeamName(teamName)

            If teamName <> "" Then

                'Extra indent spaces keep the team name clear
                'of the flag graphic drawn on top of this cell.
                target.value = "           " & teamName

                teamIndex = TeamIndexFromName(teamName)

                If teamIndex >= 0 Then

                    DrawTeamFlagGraphic _
                        ws, _
                        teamIndex, _
                        target.Left + 4, _
                        target.top + 7, _
                        24, _
                        16, _
                        "WCHighScoreFlag__" & CStr(target.Row)

                End If

            End If

        End If

    Next target

    On Error GoTo 0

End Sub

'=========================================================
' SAVE HIGH SCORE
'
' Ranking order:
'   1. Result (World Cup Champion is best,
'      then Semi-Final, Quarter-Final, Round of 16,
'      Group Stage)
'   2. Score (higher first)
'   3. Goals (more first)
'   4. Time (faster first, final tie-break)
'=========================================================

Private Sub SaveHighScore( _
    ByVal finalScore As Long, _
    ByVal finalGoals As Long, _
    ByVal finalTime As Long, _
    Optional ByVal finalResult As String = "")

    Dim ws As Worksheet

    Dim r As Long
    Dim insertRow As Long

    Dim existingScore As Long
    Dim existingGoals As Long
    Dim existingTime As Long
    Dim existingResult As String

    Set ws = ThisWorkbook.Worksheets(SHEET_NAME)

    insertRow = 13

    If finalResult = "" Then
        finalResult = "WORLD CUP CHAMPION"
    End If

    '=========================================================
    ' FIND INSERT POSITION
    '=========================================================

    For r = 3 To 12

        'Empty position
        If ws.Cells(r, 31).value = "" Then

            insertRow = r

            Exit For

        End If

        existingScore = _
            Val(ws.Cells(r, 31).value)

        existingGoals = _
            Val(ws.Cells(r, 32).value)

        existingResult = _
            CStr(ws.Cells(r, 34).value)

        If ws.Cells(r, 33).value <> "" Then

            existingTime = _
                CLng(ws.Cells(r, 33).value * 86400)

        Else

            existingTime = 999999

        End If

        '1. Better result goes first
        If ResultRank(finalResult) > ResultRank(existingResult) Then

            insertRow = r

            Exit For

        End If

        If ResultRank(finalResult) = ResultRank(existingResult) Then

            '2. Same result: higher score goes first
            If finalScore > existingScore Then

                insertRow = r

                Exit For

            End If

            '3. Same score: more goals go first
            If finalScore = existingScore Then

                If finalGoals > existingGoals Then

                    insertRow = r

                    Exit For

                End If

                '4. Same goals: faster time goes first
                If finalGoals = existingGoals Then

                    If finalTime < existingTime Then

                        insertRow = r

                        Exit For

                    End If

                End If

            End If

        End If

    Next r

    'Not good enough for top 10
    If insertRow > 12 Then Exit Sub

    '=========================================================
    ' MOVE OLD SCORES DOWN
    '=========================================================

    For r = 12 To insertRow + 1 Step -1

        ws.Cells(r, 31).value = _
            ws.Cells(r - 1, 31).value

        ws.Cells(r, 32).value = _
            ws.Cells(r - 1, 32).value

        ws.Cells(r, 33).value = _
            ws.Cells(r - 1, 33).value

        ws.Cells(r, 34).value = _
            ws.Cells(r - 1, 34).value

        ws.Cells(r, 35).value = _
            ws.Cells(r - 1, 35).value

    Next r

    '=========================================================
    ' INSERT NEW SCORE
    '=========================================================

    ws.Cells(insertRow, 31).value = finalScore

    ws.Cells(insertRow, 32).value = finalGoals

    ws.Cells(insertRow, 33).value = _
        finalTime / 86400

    ws.Cells(insertRow, 34).value = _
        finalResult

    ws.Cells(insertRow, 35).value = _
        "           " & selectedTeam

    ws.Cells(insertRow, 33).NumberFormat = "[m]:ss"

    RefreshHighScoreFlags

    '=========================================================
    ' UPDATE RANKS
    '=========================================================

    For r = 3 To 12

        ws.Cells(r, 30).value = r - 2

    Next r

End Sub

'=========================================================
' RESULT RANK
'
' Converts a result string into a ranking tier so the
' high score table can order entries by result first.
'=========================================================

Private Function ResultRank(ByVal resultText As String) As Long

    Select Case UCase$(Trim$(resultText))

        Case "WORLD CUP CHAMPION"
            ResultRank = 5

        Case "SEMI-FINAL", "SEMI FINAL", "WORLD CUP FINAL"
            ResultRank = 4

        Case "QUARTER-FINAL", "QUARTER FINAL"
            ResultRank = 3

        Case "ROUND OF 16"
            ResultRank = 2

        Case "GROUP STAGE"
            ResultRank = 1

        Case Else
            ResultRank = 0

    End Select

End Function
